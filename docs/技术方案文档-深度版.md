**run_turn 完整流程源码级解析：**

```rust
pub fn run_turn(&mut self, user_input, prompter) -> Result<TurnSummary, RuntimeError> {
    // 1. 将用户输入追加到消息历史
    self.session.messages.push(ConversationMessage::user_text(user_input));

    // 2. 主循环
    loop {
        // 2a. 构建 API 请求
        let request = ApiRequest {
            system_prompt: self.system_prompt.clone(),
            messages: self.session.messages.clone(),
        };

        // 2b. SSE 流式调用
        let events = self.api_client.stream(request)?;

        // 2c. 从 SSE 事件构建 AssistantMessage
        let (assistant_message, usage) = build_assistant_message(events)?;
        self.usage_tracker.record(usage);

        // 2d. 提取待执行工具调用列表
        let pending_tool_uses = assistant_message.blocks
            .iter()
            .filter_map(|block| match block {
                ContentBlock::ToolUse { id, name, input } => Some((id, name, input)),
                _ => None,
            })
            .collect::<Vec<_>>();

        // 2e. 追加 Assistant 消息到历史
        self.session.messages.push(assistant_message.clone());

        // 2f. 无工具调用 → 退出循环
        if pending_tool_uses.is_empty() { break; }

        // 2g. 逐个执行工具调用
        for (tool_use_id, tool_name, input) in pending_tool_uses {
            // 权限检查
            let permission_outcome = self.permission_policy.authorize(...);

            match permission_outcome {
                PermissionOutcome::Allow => {
                    // PreToolUse Hook
                    let pre_hook = self.hook_runner.run_pre_tool_use(&tool_name, &input);
                    if pre_hook.is_denied() {
                        // Hook 拒绝 → 返回拒绝消息
                        result = tool_result("PreToolUse hook denied...", true);
                    } else {
                        // 执行工具
                        let (output, is_error) = self.tool_executor.execute(&tool_name, &input)?;

                        // PostToolUse Hook
                        let post_hook = self.hook_runner.run_post_tool_use(...);

                        // 合并 Hook 反馈到输出
                        output = merge_hook_feedback(pre_hook.messages(), output, ...);
                        result = tool_result(output, is_error);
                    }
                }
                PermissionOutcome::Deny { reason } => {
                    result = tool_result(reason, true);
                }
            }

            // 追加工具结果到历史
            self.session.messages.push(result.clone());
        }
    }

    // 3. 检查是否需要自动压缩
    let auto_compaction = self.maybe_auto_compact();

    // 4. 返回 TurnSummary
    Ok(TurnSummary { ... })
}
```

### 4.3 权限系统（permissions.rs）

**PermissionPolicy 的工作方式：**

```rust
pub fn authorize(&self, tool_name: &str, input: &str,
                 prompter: Option<&mut dyn PermissionPrompter>) -> PermissionOutcome {
    let current_mode = self.active_mode();
    let required_mode = self.required_mode_for(tool_name);  // 工具要求 vs 当前模式

    // Allow 模式或当前模式 >= 要求模式 → 直接放行
    if current_mode == PermissionMode::Allow || current_mode >= required_mode {
        return PermissionOutcome::Allow;
    }

    // 否则需要交互确认
    if let Some(p) = prompter {
        let decision = p.decide(&PermissionRequest { ... });
        match decision {
            PermissionPromptDecision::Allow => PermissionOutcome::Allow,
            PermissionPromptDecision::Deny { reason } => PermissionOutcome::Deny { reason },
        }
    } else {
        PermissionOutcome::Deny { reason: "no prompter available".into() }
    }
}
```

```rust
pub fn glob_search(pattern: &str, cwd: &Path) -> Result<Vec<String>, ToolError> {
    let pattern_full = cwd.join(pattern);
    let matches = glob::glob(&pattern_full.to_string_lossy())
        .map_err(|e| ToolError::new(e.to_string()))?
        .filter_map(Result::ok)
        .map(|p| p.to_string_lossy().to_string())
        .collect();
    Ok(matches)
}
```

### 5.3 grep_search 实现
```rust
pub fn grep_search(pattern: &str, path: Option<&str>) -> Result<Vec<GrepMatch>, ToolError> {
    let re = RegexBuilder::new(pattern)
        .case_insensitive(true)
        .build()
        .map_err(|e| ToolError::new(e.to_string()))?;

    let matches: Vec<GrepMatch> = WalkDir::new(path)
        .into_iter()
        .filter_map(|e| e.ok())
        .filter(|e| e.file_type().is_file())
        .filter_map(|e| {
            let content = fs::read_to_string(e.path()).ok()?;
            let line_matches: Vec<(usize, String)> = content.lines()
                .enumerate()
                .filter(|(_, line)| re.is_match(line))
                .map(|(i, l)| (i + 1, l.to_string()))
                .collect();
            if line_matches.is_empty() { None }
            else { Some(GrepMatch { path: e.path().display().to_string(), lines: line_matches }) }
        })
        .collect();
    Ok(matches)
}
```

## 九、安全模型

### 9.1 威胁与防护

| 威胁 | 代码中的防护 |
|------|------------|
| 任意文件读取 | ReadOnly 模式下 ReadFileTool 可写文件但不可执行 Bash |
| 恶意命令注入 | BashTool 沙箱 + namespace 隔离 + 超时 |
| 路径遍历 | WorkspaceOnly 文件系统隔离 |
| Token 耗尽 | auto_compact 阈值控制 |
| API Key 泄露 | 不持久化到磁盘，仅内存持有 |
| MCP 服务恶意 | 进程隔离 + JSON-RPC 参数验证 |
| Hook 脚本滥用 | Hook 只能修改参数/结果，不能绕过权限 |
